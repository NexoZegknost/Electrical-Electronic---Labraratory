\section{Bài tập 10: Switching Power Supply With LM2596}

Bài tập khảo sát nguồn xung dùng IC hạ áp (Buck Regulator) Texas Instruments LM2596-5.0 qua 3 giai đoạn, từ mạch chưa hoàn chỉnh đến mạch có cuộn cảm và Zener. Mô phỏng Transient trong $10\text{ ms}$; các tham số \texttt{RELTOL}, \texttt{VNTOL}, \texttt{ABSTOL} được chỉnh theo hướng dẫn của đề để nguồn xung hội tụ.

\Needspace{0.4\textheight}
\subsection{Bước 1: Mạch chưa có cuộn cảm và Zener ổn áp}
\given{ex10_given}{0.85\textwidth}{Sơ đồ mạch nguồn xung chưa hoàn chỉnh với LM2596 (Hình 2.32)}

\simresult
\simfig{ex10_step1_sim}{Kết quả mô phỏng điện áp ngõ ra tại Bước 1}

\begin{notebox}{Nhận xét, Bước 1}
  \begin{itemize}[leftmargin=1.5em,itemsep=2pt,topsep=2pt]
    \item Số liệu đọc bằng con trỏ: điện áp ngõ ra bằng $0\text{ V}$ trong khoảng $0.9\text{ ms}$ đầu, tăng lên đỉnh $V_{\text{peak}} = 5.6515\text{ V}$ tại $t = 3.892\text{ ms}$ (vọt lố khoảng $0.65\text{ V}$, tức $13\%$), giảm xuống đáy $V_{\text{dip}} = 4.8336\text{ V}$ tại $t = 5.318\text{ ms}$, rồi ổn định quanh $5.0\text{ V}$ từ khoảng $5.5\text{ ms}$.
    \item Vấn đề của ngõ ra: điện áp không lên $5\text{ V}$ một cách êm mà bị vọt lố rồi mới ổn định.
    \item Nguyên nhân: chân \texttt{OUTPUT} của LM2596 là nút chuyển mạch, xuất xung đóng cắt khoảng $150\text{ kHz}$ có biên độ cỡ điện áp vào. Không có cuộn cảm, các xung này nạp thẳng vào $C_2$ và tải mà dòng không được giới hạn. Tín hiệu hồi tiếp (\texttt{FEEDBACK}) lấy trực tiếp từ nút xung nên vòng điều khiển phản ứng trễ.
    \item Chưa có Zener ghim áp nên đỉnh vọt lố ($5.65\text{ V} > 5\text{ V}$) truyền thẳng ra tải và có thể gây quá áp.
  \end{itemize}
\end{notebox}

\Needspace{0.4\textheight}
\subsection{Bước 2: Thêm cuộn cảm \texorpdfstring{$L_1 = 33\,\mu\text{H}$}{L1 = 33uH}}
\given{ex10_step2_given}{0.85\textwidth}{Mạch thêm cuộn cảm $L_1 = 33\,\mu\text{H}$ (Hình 2.35)}

\simresult
\simfig{ex10_step2_sim}{Kết quả mô phỏng điện áp ngõ ra sau khi thêm cuộn cảm $L_1 = 33\,\mu\text{H}$}

\begin{notebox}{Nhận xét, Bước 2}
  \begin{itemize}[leftmargin=1.5em,itemsep=2pt,topsep=2pt]
    \item Số liệu đọc bằng con trỏ: điện áp bắt đầu tăng sau khoảng $0.9\text{ ms}$ và đạt $5\text{ V}$ sau khoảng $0.9\text{ ms}$ nữa; đỉnh $V_{\text{peak}} = 5.0745\text{ V}$ tại $t = 1.812\text{ ms}$ (vọt lố $0.07\text{ V}$, tức $1.5\%$); đáy $V_{\text{dip}} = 4.9624\text{ V}$ tại $t = 2.045\text{ ms}$, sau đó ổn định quanh $5.0\text{ V}$.
    \item Cải thiện so với Bước 1: vọt lố giảm từ $5.65\text{ V}$ xuống $5.07\text{ V}$, thời gian ổn định rút từ khoảng $5.5\text{ ms}$ xuống khoảng $2\text{ ms}$.
    \item Cuộn cảm chống lại sự thay đổi đột ngột của dòng điện ($v_L = L\,\dfrac{di}{dt}$). Khi công tắc trong IC đóng, cuộn cảm tích năng lượng và hạn chế dòng tăng vọt, làm mịn dòng nạp vào tụ và tải.
    \item Ngõ ra vẫn còn vọt lố nhỏ khi khởi động và chưa có linh kiện ghim áp để bảo vệ tải khỏi quá áp.
  \end{itemize}
\end{notebox}

\subsection{Bước 3: Thêm Diode Zener \texorpdfstring{$5\text{ V}$}{5V} và đổi tụ \texorpdfstring{$C_2 = 220\,\mu\text{F}$}{C2 = 220uF}}
\given{ex10_step3_given}{0.85\textwidth}{Mạch hoàn chỉnh với Diode Zener và tụ $C_2 = 220\,\mu\text{F}$ (Hình 2.36)}

\simresult
\simfig{ex10_step3_sim}{Kết quả mô phỏng điện áp ngõ ra và dòng qua Diode Zener tại Bước 3}

\begin{notebox}{Nhận xét, Bước 3}
  \begin{itemize}[leftmargin=1.5em,itemsep=2pt,topsep=2pt]
    \item Số liệu đọc bằng con trỏ: điện áp đạt $5\text{ V}$ sau khoảng $2\text{ ms}$; đỉnh $V_{\text{peak}} = 5.0560\text{ V}$ tại $t = 2.102\text{ ms}$ (vọt lố $0.06\text{ V}$, tức $1.1\%$, nhỏ hơn mức $5.0745\text{ V}$ ở Bước 2); giá trị xác lập $V_{\text{out}} = 4.9915\text{ V}$ tại $t = 9.5\text{ ms}$.
    \item Dòng qua Zener $I(D_5{:}2)$ chỉ cỡ femtoampe ($-59.8\text{ fA}$ và $-147.5\text{ fA}$), tức $I_Z \approx 0$: Zener $5\text{ V}$ gần như không dẫn trong suốt quá trình mô phỏng.
    \item Vai trò của Zener $D_5$ (mắc ngược từ ngõ ra xuống mass, song song với tải): ghim điện áp ngõ ra ở mức $V_Z = 5\text{ V}$. Khi điện áp có xu hướng vượt $5\text{ V}$ (do gai hoặc gợn sóng của xung đóng cắt), Zener dẫn dòng xuống mass để giữ điện áp tải và bảo vệ tải khỏi quá áp. Trong mô phỏng này ngõ ra chỉ vọt lên tối đa $5.056\text{ V}$ nên Zener chỉ đóng vai trò dự phòng.
    \item Tụ $C_2 = 220\,\mu\text{F}$ kết hợp với $L_1$ tạo bộ lọc thông thấp LC, làm trơn thành phần xung $\approx 150\text{ kHz}$ ở ngõ ra.
  \end{itemize}
\end{notebox}
