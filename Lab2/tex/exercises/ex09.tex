\section{Bài tập 9: AC/DC Power Circuit Application}

Trong bài tập này, chúng ta sẽ từng bước khảo sát quá trình xây dựng một bộ nguồn chuyển đổi từ điện áp xoay chiều (AC) sang điện áp một chiều (DC) ổn định qua 5 bước thực nghiệm. Mỗi bước sẽ được tiến hành mô phỏng trong miền thời gian (Transient analysis), kèm theo đồ thị và lời giải thích chi tiết về vai trò của từng linh kiện.

Nguồn xoay chiều đầu vào $V_1$ có các thông số: \texttt{VOFF} = $0\text{ V}$, \texttt{VAMPL} = $220\text{ V}$, \texttt{FREQ} = $50\text{ Hz}$, \texttt{AC} = $1$, biến áp $TX_1$ tỷ số $10:1$ và cầu chỉnh lưu 4 diode \texttt{Dbreak}.

\subsection{Bước 1: Điện áp chỉnh lưu chưa qua mạch lọc hoặc ổn áp}
Mạch gồm cuộn thứ cấp biến áp kết nối với cầu chỉnh lưu diode và cấp trực tiếp cho điện trở tải $R_1 = 1\text{ k}\Omega$ (chưa có tụ lọc hay diode Zener).

\given{ex9_given}{0.65\textwidth}{Sơ đồ mạch Bước 1: Mạch chỉnh lưu chưa có mạch lọc (Hình 2.29)}

\simresult
\simfig{ex9_step1_sim}{Kết quả mô phỏng dạng sóng điện áp ngõ ra tại Bước 1}

\begin{notebox}{Quan sát và số liệu đọc từ mô phỏng, Bước 1}
  \begin{itemize}[leftmargin=1.5em]
    \item Điện áp ngõ ra $V(R1{:}2)$ là sóng một chiều nhấp nhô, các đỉnh cách nhau $10\text{ ms}$ (tần số gợn sóng $100\text{ Hz}$), điện áp luôn $\ge 0$.
    \item Đỉnh đọc bằng con trỏ: $V_{\max} = 21.105\text{ V}$ (tại $t = 15\text{ ms}$).
    \item Đáy đọc bằng con trỏ: $V_{\min} \approx 0.156\text{ V} \approx 0\text{ V}$ (tại $t \approx 30\text{ ms}$), gợn sóng $V_r \approx 20.95\text{ V}$.
  \end{itemize}
\end{notebox}

\noindent\textbf{Giải thích và nhận xét:}
\begin{itemize}[leftmargin=1.5em]
  \item Dạng sóng điện áp trên tải $R_1$ là điện áp một chiều nhấp nhô hoàn toàn (pulsating DC) với tần số gợn sóng $f_{\text{ripple}} = 2f = 100\text{ Hz}$ và chu kỳ $T = 10\text{ ms}$.
  \item Biên độ điện áp đỉnh đạt xấp xỉ $V_p = \frac{220}{10} - 2 \times 0.7\text{ V} = 20.6\text{ V}$. Tuy nhiên, ở cuối mỗi nửa chu kỳ, điện áp rơi về $0\text{ V}$. Dạng sóng này chưa thể sử dụng trực tiếp để cấp nguồn cho các thiết bị điện tử.
\end{itemize}

\subsection{Bước 2: Mạch chỉnh lưu có tụ lọc nhỏ \texorpdfstring{$C_1 = 10\,\mu\text{F}$}{C1 = 10uF}}
Mắc song song một tụ điện dung lượng nhỏ $C_1 = 10\,\mu\text{F}$ với điện trở tải $R_1 = 1\text{ k}\Omega$.

\given{ex9_step2_given}{0.65\textwidth}{Sơ đồ mạch Bước 2: Thêm tụ lọc $C_1 = 10\,\mu\text{F}$ (Hình 2.30)}

\simresult
\simfig{ex9_step2_sim}{Kết quả mô phỏng dạng sóng ngõ ra tại Bước 2 ($C_1 = 10\,\mu\text{F}$)}

\begin{notebox}{Quan sát và số liệu đọc từ mô phỏng, Bước 2}
  \begin{itemize}[leftmargin=1.5em]
    \item Điện áp ngõ ra không còn về $0\text{ V}$ mà có dạng răng cưa: tăng nhanh theo đỉnh chỉnh lưu (tụ nạp), giảm dần giữa hai đỉnh (tụ xả qua $R_1$).
    \item Đỉnh đọc bằng con trỏ: $V_{\max} = 21.110\text{ V}$ (tại $t = 15\text{ ms}$).
    \item Đáy đọc bằng con trỏ: $V_{\min} = 10.836\text{ V}$ (tại $t \approx 31.5\text{ ms}$), gợn sóng $V_r \approx 10.27\text{ V}$ còn rất lớn.
  \end{itemize}
\end{notebox}

\noindent\textbf{Giải thích và nhận xét:}
\begin{itemize}[leftmargin=1.5em]
  \item \textbf{Cơ chế nạp - xả của tụ điện:} Khi điện áp chỉnh lưu tăng từ $0\text{ V}$ lên đỉnh ($20.6\text{ V}$), tụ điện $C_1$ được nạp nhanh qua các diode. Khi điện áp chỉnh lưu giảm xuống dưới điện áp trên tụ, các diode bị phân cực ngược và ngắt, tụ $C_1$ phóng điện qua tải $R_1$ để duy trì điện áp cấp cho tải.
  \item Tuy nhiên, do dung lượng tụ khá nhỏ ($C_1 = 10\,\mu\text{F}$), hằng số thời gian phóng điện $\tau = R_1 C_1 = 1000 \times 10 \times 10^{-6} = 10\text{ ms}$ chỉ ngang với nửa chu kỳ tín hiệu ($10\text{ ms}$). Do đó, tụ phóng điện nhanh và biên độ gợn sóng (ripple voltage) vẫn còn rất lớn. Công thức gần đúng $V_r \approx I_{dc}/(2fC_1)$ chỉ đúng khi gợn sóng nhỏ, nên ở đây dùng mô hình xả mũ $v_C(t) = V_p\,e^{-(t-t_0)/\tau}$: tụ xả khoảng $8\text{ ms}$ giữa hai đỉnh nên điện áp trên tụ giảm xuống còn khoảng $V_{\min} \approx 10.8\text{ V}$, tức gợn sóng:
  \begin{equation}
    V_r = V_{\max} - V_{\min} \approx 20.6 - 10.8 \approx 9.8\text{ V}
  \end{equation}
\end{itemize}

\subsection{Bước 3: Tăng dung lượng tụ lọc lên \texorpdfstring{$C_1 = 680\,\mu\text{F}$}{C1 = 680uF}}
Thay thế tụ $10\,\mu\text{F}$ bằng tụ có dung lượng lớn $C_1 = 680\,\mu\text{F}$.

\simresult
\simfig{ex9_step3_sim}{Kết quả mô phỏng dạng sóng ngõ ra tại Bước 3 ($C_1 = 680\,\mu\text{F}$)}

\begin{notebox}{Quan sát và số liệu đọc từ mô phỏng, Bước 3}
  \begin{itemize}[leftmargin=1.5em]
    \item Sau khoảng $15\text{ ms}$ đầu (giai đoạn tụ nạp), điện áp ngõ ra gần như phẳng ở mức xấp xỉ $20.5\text{ V}$; gợn sóng nhỏ hơn rất nhiều so với Bước 2.
    \item Đọc bằng con trỏ trong một chu kỳ ổn định ($t = 75\text{ ms}$ đến $85\text{ ms}$): $V_{\max} = 20.684\text{ V}$, $V_{\min} = 20.420\text{ V}$.
    \item Gợn sóng $V_r = 20.684 - 20.420 \approx 0.263\text{ V}$.
  \end{itemize}
\end{notebox}

\noindent\textbf{Giải thích và nhận xét:}
\begin{itemize}[leftmargin=1.5em]
  \item Khi dung lượng tụ tăng lên $680\,\mu\text{F}$, hằng số thời gian phóng điện tăng mạnh:
  \begin{equation}
    \tau = R_1 C_1 = 1000 \times 680 \times 10^{-6} = 0.68\text{ s} = 680\text{ ms} \gg 10\text{ ms}
  \end{equation}
  \item Tụ điện phóng điện rất chậm trong khoảng thời gian giữa hai đỉnh sóng liên tiếp. Độ gợn sóng (ripple) giảm đi đáng kể:
  \begin{equation}
    V_r \approx \frac{I_{dc}}{2 f C_1} \approx \frac{20\text{ mA}}{100 \times 680 \times 10^{-6}} \approx 0.29\text{ V}
  \end{equation}
  \item Điện áp ngõ ra hầu như được làm phẳng hoàn toàn, duy trì mức điện áp một chiều gần như không đổi, dao động trong khoảng $20.3\text{ V}$ đến $20.6\text{ V}$.
\end{itemize}

\subsection{Bước 4: Mắc thêm Diode Zener ổn áp (\texorpdfstring{$V_Z = 22\text{ V}$}{VZ = 22V})}
Mắc thêm một diode Zener parameterized (\texttt{Zener\_P}) song song với tụ lọc $C_1$ và tải $R_1$, với thuộc tính điện áp Zener được đặt là \texttt{ZENER\_VOLTAGE} = $22\text{ V}$.

\given{ex9_step4_given}{0.65\textwidth}{Sơ đồ mạch Bước 4: Mạch chỉnh lưu có tụ lọc và Diode Zener (Hình 2.31)}

\simresult
\simfig{ex9_step4_sim}{Kết quả mô phỏng dạng sóng ngõ ra và dòng Zener tại Bước 4 ($V_Z = 22\text{ V}$)}

\begin{notebox}{Quan sát và số liệu đọc từ mô phỏng, Bước 4}
  \begin{itemize}[leftmargin=1.5em]
    \item Dạng sóng ngõ ra $V(D5{:}2)$ giống Bước 3: phẳng ở mức xấp xỉ $20.5\text{ V}$ sau giai đoạn nạp ban đầu.
    \item Đọc bằng con trỏ trong chu kỳ ổn định ($t \approx 94\text{--}95\text{ ms}$): $V_{\max} = 20.652\text{ V}$, $V_{\min} = 20.146\text{ V}$, gợn sóng $V_r \approx 0.506\text{ V}$.
    \item Dòng qua Zener $I(D5{:}2) \approx 0.12\text{ mA}$ ($121\text{--}124\,\mu\text{A}$), xấp xỉ $0$ trong suốt thời gian mô phỏng: Zener $22\text{ V}$ chưa dẫn.
  \end{itemize}
\end{notebox}

\noindent\textbf{Giải thích và nhận xét:}
\begin{itemize}[leftmargin=1.5em]
  \item Điện áp một chiều sau mạch lọc ở Bước 3 có đỉnh khoảng $20.6\text{ V}$ và thấp nhất khoảng $20.3\text{ V}$.
  \item Do điện áp trên tụ ($\le 20.6\text{ V}$) \textbf{nhỏ hơn} điện áp đánh thủng ngược của diode Zener ($V_Z = 22\text{ V}$), diode Zener hoàn toàn chưa dẫn điện (dòng qua Zener xấp xỉ bằng $0$: $I_Z \approx 0$).
  \item Do đó, diode Zener ở bước này chưa tham gia vào quá trình điều chỉnh điện áp, hình dạng sóng đầu ra hoàn toàn giống như ở Bước 3.
\end{itemize}

\subsection{Bước 5: Đổi điện áp Diode Zener thành \texorpdfstring{$V_Z = 20\text{ V}$}{VZ = 20V}}
Thay đổi thuộc tính điện áp Zener thành \texttt{ZENER\_VOLTAGE} = $20\text{ V}$ và chạy lại mô phỏng.

\simresult
\simfig{ex9_step5_sim}{Kết quả mô phỏng dạng sóng ngõ ra và dòng Zener tại Bước 5 ($V_Z = 20\text{ V}$)}

\begin{notebox}{Quan sát và số liệu đọc từ mô phỏng, Bước 5}
  \begin{itemize}[leftmargin=1.5em]
    \item Điện áp ngõ ra $V(R1{:}2)$ bị ghim quanh $20\text{ V}$ (thấp hơn mức $\approx 20.5\text{ V}$ ở Bước 4).
    \item Đọc bằng con trỏ trong chu kỳ ổn định ($t \approx 94\text{--}95\text{ ms}$): $V_{\max} = 20.081\text{ V}$, $V_{\min} = 19.746\text{ V}$, gợn sóng $V_r \approx 0.334\text{ V}$ (nhỏ hơn $0.506\text{ V}$ ở Bước 4).
    \item Dòng qua Zener xuất hiện thành các xung dẫn tại các đỉnh điện áp ($t \approx 15, 35, 55, 75, 95\text{ ms}$), đỉnh xung đọc được $I_Z \approx 2.21\text{ A}$ tại $t = 94.9\text{ ms}$; ngoài xung, $I_Z \approx 0.13\text{ mA}$.
  \end{itemize}
\end{notebox}

\noindent\textbf{Giải thích và nhận xét:}
\begin{itemize}[leftmargin=1.5em]
  \item Vì điện áp đỉnh của tụ lọc ($20.3\text{ V}$ đến $20.6\text{ V}$) \textbf{lớn hơn} điện áp đánh thủng Zener ($V_Z = 20\text{ V}$), diode Zener lập tức bị đánh thủng ngược và ghim điện áp đầu ra quanh mức $20\text{ V}$.
  \item Dòng điện dư thừa và độ gợn sóng còn sót lại trên tụ lọc được dẫn xả qua diode Zener xuống đất (dòng $I_Z > 0$). Nhờ đó, dạng sóng đầu ra gần như phẳng ở mức xấp xỉ $20\text{ V}$ (còn sai lệch nhỏ do nội trở động của Zener thực tế), gợn sóng giảm còn khoảng $0.33\text{ V}$ (so với $0.51\text{ V}$ ở Bước 4).
\end{itemize}
