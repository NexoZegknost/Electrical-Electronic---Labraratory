\section{Exercise 6}

Given the following circuit. Apply the knowledge you've learned to transform it into another form in which you can find total equivalent resistance $R_{ab}$ more easily. Next, find the value of the current $i$ through the circuit and perform a simulation to check it out.

\given{ex6_given}{0.55\textwidth}{Transform the circuit, then find the equivalent resistance $R_{ab}$ and the current $i$}

\subsection{Circuit transformation}
\drawfig{ex6_transform}{Transformed circuit}
\noindent Cấu trúc cầu gồm 5 điện trở $24\,\Omega, 10\,\Omega, 20\,\Omega, 30\,\Omega, 50\,\Omega$ không cân bằng ($24 \times 50 \neq 10 \times 30$). Đặt tên các nút $T$ (trên), $M$ (giữa trái), $P$ (giữa phải) và $b$ (dưới). Biến đổi tam giác ($\Delta$) $T$-$M$-$P$ gồm $R_{TM} = 24\,\Omega, R_{TP} = 10\,\Omega, R_{MP} = 20\,\Omega$ thành hình sao (Y) với tâm sao $N$:
\[
\Sigma_\Delta = R_{TM} + R_{TP} + R_{MP} = 24 + 10 + 20 = 54\,\Omega
\]
\begin{align*}
R_1 &= \frac{R_{TM} \cdot R_{TP}}{\Sigma_\Delta} = \frac{24 \times 10}{54} = \frac{40}{9}\,\Omega \approx 4{,}444\,\Omega \quad (\text{nối vào nút } T) \\
R_2 &= \frac{R_{TM} \cdot R_{MP}}{\Sigma_\Delta} = \frac{24 \times 20}{54} = \frac{80}{9}\,\Omega \approx 8{,}889\,\Omega \quad (\text{nối vào nút } M) \\
R_3 &= \frac{R_{TP} \cdot R_{MP}}{\Sigma_\Delta} = \frac{10 \times 20}{54} = \frac{100}{27}\,\Omega \approx 3{,}704\,\Omega \quad (\text{nối vào nút } P)
\end{align*}

\subsection{Calculation}
\notes
\begin{enumerate}[leftmargin=2em]
  \item \textbf{Gộp nối tiếp hai nhánh từ $N$ xuống $b$:}
        \begin{align*}
        R_{Nb1} &= R_2 + 30\,\Omega = \frac{80}{9} + 30 = \frac{350}{9}\,\Omega \approx 38{,}889\,\Omega \\
        R_{Nb2} &= R_3 + 50\,\Omega = \frac{100}{27} + 50 = \frac{1450}{27}\,\Omega \approx 53{,}704\,\Omega
        \end{align*}
  \item \textbf{Gộp hai nhánh song song giữa $N$ và $b$:}
        \[
        R_{Nb} = R_{Nb1} \parallel R_{Nb2} = \frac{\frac{350}{9} \times \frac{1450}{27}}{\frac{350}{9} + \frac{1450}{27}} = \frac{203}{9}\,\Omega \approx 22{,}556\,\Omega
        \]
  \item \textbf{Điện trở tương đương toàn mạch $R_{ab}$:}
        \[
        R_{ab} = 13\,\Omega + R_1 + R_{Nb} = 13 + \frac{40}{9} + \frac{203}{9} = 13 + \frac{243}{9} = 13 + 27 = 40\,\Omega
        \]
  \item \textbf{Dòng điện $i$ qua toàn mạch:}
        \[
        i = \frac{100\text{ V}}{R_{ab}} = \frac{100\text{ V}}{40\,\Omega} = 2{,}5\text{ A}
        \]
\end{enumerate}

\noindent\textbf{Tổng hợp kết quả:}
\begin{itemize}[leftmargin=2em]
  \item $R_{ab} = 40\,\Omega$
  \item $i = 2{,}5\text{ A}$
\end{itemize}
