\section{Bài tập 1}

Cho mạch điện như hình bên dưới. Hãy tính giá trị của điện áp $v_0$ và dòng điện $i$. Sau đó, mô phỏng mạch điện để kiểm tra lại kết quả.

\given{ex1_given}{0.5\textwidth}{Tìm điện áp và dòng điện trong mạch đã cho bằng KVL}

\subsection{Tính toán}
\noindent\textbf{Theo định luật Ohm trên điện trở $6\,\Omega$:} Dòng $i$ đi vào cực âm của $v_0$ (cực $+$ ở bên trái):
\begin{equation}
v_0 = -6i \label{eq:ex1_1}
\end{equation}
\noindent\textbf{Áp dụng định luật Kirchhoff về điện áp (KVL)} theo chiều kim đồng hồ bắt đầu từ góc dưới bên trái:
\begin{equation}
-12\text{ V} + 4i + 2v_0 - 4\text{ V} + 6i = 0 \iff 10i + 2v_0 = 16 \label{eq:ex1_2}
\end{equation}
\noindent\textbf{Từ (1) và (2):} Thay $v_0 = -6i$ vào phương trình KVL:
\begin{align*}
10i + 2(-6i) = 16 &\implies -2i = 16 \implies i = -8\text{ A} \\
&\implies v_0 = -6 \cdot (-8\text{ A}) = 48\text{ V}
\end{align*}

\begin{dapsobox}
  \begin{itemize}[leftmargin=1.5em]
    \item $v_0 = 48\text{ V}$
    \item $i = -8\text{ A}$ \quad (dòng điện thực tế có độ lớn $8\text{ A}$ chạy ngược chiều kim đồng hồ)
  \end{itemize}
\end{dapsobox}
