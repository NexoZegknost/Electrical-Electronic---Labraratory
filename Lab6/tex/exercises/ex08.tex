\subsection{MOSFET Motor Controller \bonus}\label{sec:motor}
\given{fig6_15}{0.6\linewidth}{Simple motor controller using EFET\label{fig:6-15}}

\subsubsection*{Nguyên lý}
\begin{itemize}[nosep]
  \item Động cơ là tải cảm. Khi MOSFET tắt, cuộn dây sinh sức phản điện động (back-emf) $v_L = L\,\dfrac{di}{dt}$ rất lớn; diode flywheel song song với động cơ dẫn dòng này về lại nguồn, bảo vệ MOSFET.
  \item Mạng snubber/clamping (zener nối tiếp diode) cho phép dòng tắt nhanh hơn và giới hạn đỉnh điện áp ngược.
  \item $D_1$ đặt ngang kênh D--S bảo vệ MOSFET khỏi quá áp chuyển mạch; $R_{GS}$ kéo cổng về $0\,\text{V}$ khi ngõ TTL ở mức thấp.
\end{itemize}

\given{fig6_16}{0.6\linewidth}{Full Motor controller with N and P channel EFET\label{fig:6-16}}

\subsubsection*{Nguyên lý}
\begin{itemize}[nosep]
  \item $V_{IN}$ thấp: P-MOSFET ($\text{FET}_1$) có $V_{GS} < 0$ nên dẫn, N-MOSFET tắt; động cơ nối với $+V_{DD}$ và quay thuận (high-side switching).
  \item $V_{IN}$ cao: P-MOSFET tắt, N-MOSFET dẫn; động cơ nối với $-V_{DD}$, điện áp hai đầu đảo chiều nên động cơ quay ngược (low-side switching).
\end{itemize}
